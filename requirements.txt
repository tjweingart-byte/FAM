anthropic>=1.2.0
# No speech engine here on purpose. FAM's production voice is Chatterbox,
# which needs torch and is installed separately by
# `pip install -r requirements-chatterbox.txt` - gigabytes, and only useful on
# a machine with a GPU. A laptop, a CI runner and the preview build must all be
# able to install FAM and run its tests without a deep-learning stack; there,
# `/api/health` reports `interim: true` and a placeholder tone plays.
#
# Piper used to sit here as an interim voice and is gone. A second engine that
# can speak is a second engine that can be selected, and an app that quietly
# sounds worse than intended is the failure this project has lost the most time
# to. A tone cannot be mistaken for FAM.
fastapi>=0.110
uvicorn[standard]>=0.27
# app.py imports BaseModel and Field directly. fastapi happens to pull pydantic
# in, so this has always been present by luck rather than by declaration - and
# a dependency that is only there by luck is one nobody notices losing.
pydantic>=2.0
# The same, for starlette: app.py edits response headers directly in its
# plain-ASGI middleware (§238, `Headers`/`MutableHeaders`). fastapi pulls it
# in; declared so it is not here by luck. No upper bound - fastapi sets that.
starlette>=0.27

# Not part of the deep-learning stack, and needed without it. `pcm_from_float`
# converts a float waveform to 16-bit PCM on the synthesis path, and the engine
# contract tests exercise that path with chatterbox and torch stubbed out - so
# the one environment that has neither still needs this. It was undeclared, and
# CI caught it the only way an undeclared dependency is ever caught: six tests
# failing with ModuleNotFoundError on a machine that did not happen to have it.
#
# Not replaceable with the stdlib here: a 3-minute episode at 24 kHz is ~4.3M
# samples, and converting those one at a time in Python would put seconds on
# the synthesis path.
numpy>=1.24

# Development only
pytest>=8.0

# No longer development-only. `remote_voice.py` uses this to reach a Chatterbox
# worker on a GPU somewhere else, so with VOICE_BACKEND=remote it is on the
# synthesis path of every episode - a CPU deployment that skipped it would
# install cleanly and then be unable to speak. It is also what
# fastapi.testclient runs on.
#
# The Anthropic SDK uses httpx2, which it pulls in itself - do not add httpx
# expecting the SDK to use it.
httpx>=0.27

# Kept audio is packed as Opus at 24 kbps (§237, `audio_codec.py`): about a
# twelfth of zlib's size for speech, which is what makes the R2 bucket cheap.
# PyAV's wheel carries its own libopus, so the image needs no ffmpeg. ~35 MB
# resident once used, against the starter plan's 512 MB. Optional in the
# pypdf sense: without it audio is kept as zlib and `/api/health` says why.
av>=12.0

# Postgres, for option B (§243, `db.py`): with DATABASE_URL set every store
# lives in Postgres instead of a SQLite file. Imported only then, so a laptop
# without a database never loads it. The binary wheel carries libpq: the image
# needs no system package.
psycopg[binary]>=3.2

# Optional: PDF attachments. Everything else (docx, text, links, photos)
# needs no extra package; without this a PDF fails with a message saying so.
pypdf>=4.0

# The trending bank schedules its editions at 05:00 and 17:00 in
# America/New_York (§139), through `zoneinfo`. Python ships the code and not
# the zone data; a slim container image has no /usr/share/zoneinfo, and
# without this the schedule falls back to UTC - 5am Eastern becoming 1am.
tzdata>=2024.1

# Autocorrect for messages and search (§142, `autocorrect.py`). Pure Python
# with its word list inside the wheel - no model, no network. Optional in the
# same way pypdf is: without it every word passes through unchanged and
# `/api/spell` says `available: false`.
pyspellchecker>=0.8

# Tile pictures (§160, `thumbnails.py`): crops and scales a painted picture to
# a 480x360 WebP (~30 KB) before it is stored, and a larger JPEG for the
# checker. Optional in the pypdf sense: without it the original PNG is kept
# as it came (larger, still correct) and a warning says so. Only the machine
# that paints pictures needs it; serving them needs nothing.
Pillow>=10.0

# The finance workbook (`financials.py`, `/api/admin/financials.xlsx`): every
# service FAM pays for and what it spent each day, as an Excel file built on
# request. Only the admin download imports it.
openpyxl>=3.1
